## 一、结构体定义

```cpp
struct Student {
```

## 二、已提供的函数（可直接使用）

排序功能已由 `SortByScore` 函数提供，你只需要**调用它**即可。它使用冒泡排序算法按成绩从低到高排序。

```cpp
// 按成绩升序排序（冒泡排序）
// 参数：arr - 学生数组首地址，count - 学生人数
// 使用方式：SortByScore(students, 3);
// 注意：排序时需要交换两个学生的成绩，你需要自己实现 SwapScore 函数
void SortByScore(Student* arr, int count) {
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - 1 - i; j++) {
            // 如果当前学生成绩比下一个学生高，则交换
            if ((arr + j)->score > (arr + j + 1)->score) {
                // ★ 你需要自己实现 SwapScore 函数来完成这里的交换
                SwapScore(arr + j, arr + j + 1);
            }
        }
    }
}
```

> **提示**：`SortByScore` 函数中调用了 `SwapScore` 函数，你需要自己实现它——交换两个学生的成绩。


## 三、题目要求

### 第 1 问：指针基础

在 `main` 函数中完成以下操作：

1. 定义一个 `Student` 类型的变量 `s1`，初始化为 `{1, "Alice", 95.5}`。

4. 使用两种方式（`.` 和 `->`）分别打印 `s1` 的所有成员，验证修改结果。

### 第 2 问：指针与数组

在 `main` 函数中继续完成：

1. 定义一个 `Student` 类型的数组 `students`，长度为 `3`，用列表初始化赋初值（数据自定）。

### 第 3 问：实现交换函数并排序

### 第 4 问：空指针检查

在 `main` 中分别传入 `&s1` 和 `nullptr` 测试该函数。

## 七、完整参考答案

```cpp
#include <iostream>
#include <string>
using namespace std;

struct Student {
    int id = 0;
    string name = "";
    double score = 0;
};

// ============================================================
// 已提供的排序函数
// 按成绩升序排序（冒泡排序）
// 参数：arr - 学生数组首地址，count - 学生人数
// 使用方式：SortByScore(students, 3);
// 注意：排序时需要交换两个学生的成绩，你需要自己实现 SwapScore 函数
// ============================================================
void SortByScore(Student* arr, int count) {
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - 1 - i; j++) {
            // 如果当前学生成绩比下一个学生高，则交换
            if ((arr + j)->score > (arr + j + 1)->score) {
                // ★ 调用你自己实现的 SwapScore 函数
                SwapScore(arr + j, arr + j + 1);
            }
        }
    }
}

// ============================================================
// 第 3 问：你自己实现的交换函数
// ============================================================
void SwapScore(Student* a, Student* b) {
    double temp = a->score;
    a->score = b->score;
    b->score = temp;
}

// ============================================================
// 第 4 问：打印函数（带空指针检查）
// ============================================================
void PrintStudent(const Student* p) {
    if (p == nullptr) {
        cout << "指针为空，无法打印" << endl;
    } else {
        cout << "学号：" << p->id
             << "，姓名：" << p->name
             << "，成绩：" << p->score << endl;
    }
}

// ============================================================
// main 函数
// ============================================================
int main() {
    // ===== 第 1 问 =====
    Student s1 = {1, "Alice", 95.5};
    Student* p1 = &s1;

    p1->score = 98.0;

    cout << "=== 第 1 问：通过指针访问结构体 ===" << endl;
    cout << "s1.name = " << s1.name << endl;
    cout << "s1.score = " << s1.score << endl;
    cout << "p1->name = " << p1->name << endl;
    cout << "p1->score = " << p1->score << endl;
    cout << endl;

    // ===== 第 2 问 =====
    Student students[3] = {
        {1, "Alice", 98.0},
        {2, "Bob", 72.5},
        {3, "Charlie", 88.0}
    };
    Student* p2 = students;

    cout << "=== 第 2 问：指针遍历数组 ===" << endl;
    for (int i = 0; i < 3; i++) {
        cout << "students[" << i << "]: "
             << (p2 + i)->id << ", "
             << (p2 + i)->name << ", "
             << (p2 + i)->score << endl;
    }

    Student* maxPtr = p2;
    for (int i = 1; i < 3; i++) {
        if ((p2 + i)->score > maxPtr->score) {
            maxPtr = p2 + i;
        }
    }
    cout << "最高分学生：" << maxPtr->name << "（" << maxPtr->score << "分）" << endl;
    cout << endl;

    // ===== 第 3 问 =====
    cout << "=== 第 3 问：排序 ===" << endl;
    cout << "排序前：" << endl;
    for (int i = 0; i < 3; i++) {
        cout << students[i].name << "：" << students[i].score << endl;
    }

    SortByScore(students, 3);

    cout << "排序后（按成绩升序）：" << endl;
    for (int i = 0; i < 3; i++) {
        cout << students[i].name << "：" << students[i].score << endl;
    }
    cout << endl;

    // ===== 第 4 问 =====
    cout << "=== 第 4 问：空指针检查 ===" << endl;
    PrintStudent(&s1);
    PrintStudent(nullptr);

    return 0;
}
```
